**节点实现**: `backend/src/services/mas/agents/memoryNode.ts:69-257`

```typescript
export const memoryNode = async (state: AgentStateType): Promise<Partial<AgentStateType>> => {
  const { userId, messages } = state;

  // 1. 加载现有画像
  const existingInsight = await MemoryPersistence.getInsight(userId);

  // 2. AI 提取新洞察 (Structured Output)
  const memoryData = await structuredModel.invoke([
    new SystemMessage(MEMORY_SYSTEM_PROMPT),
    new HumanMessage(extractionPrompt)
  ]);

  // 3. 合并负荷锚点
  const mergedAnchors = mergeLoadAnchors(
    existingInsight?.loadAnchors,
    memoryData.loadAnchors
  );

  // 4. 通过 ProfileUpdateManager 更新
  await profileUpdateManager.update({
    userId,
    updateType: 'load_anchors',
    data: { load_anchors: mergedAnchors },
    modifiedBy: 'mas',
    changeReason: 'Memory node: load anchor merge from AI extraction'
  });

  // 5. 更新历史摘要
  await profileUpdateManager.update({
    userId,
    updateType: 'history_summary',
    data: { recent_summary: memoryData.summary },
    modifiedBy: 'mas',
    changeReason: 'Memory node: summary update from conversation'
  });

  return {
    uiHint: {
      type: 'PROFILE_UPDATE_CONFIRM',
      data: { updatedFields: ['load_anchors', 'recent_summary'] }
    }
  };
};
```

### 阶段 6: ProfileUpdateManager 协调

**更新管理器**: `backend/src/services/mas/services/ProfileUpdateManager.ts`

```typescript
async update(params: ProfileUpdateParams): Promise<UpdateResult> {
  const { userId, updateType, data, modifiedBy, changeReason } = params;

  // 1. 获取当前状态
  const current = await this.getCurrentProfile(userId);

  // 2. 数据验证
  const validated = await this.validate(updateType, data);

  // 3. 事务处理
  return await this.db.transaction(async (tx) => {
    // 3.1 更新数据库
    const updated = await this.executeUpdate(tx, userId, validated, current);

    // 3.2 记录审计日志
    await this.auditLog(tx, {
      userId,
      updateType,
      modifiedBy,
      changeReason,
      before: current,
      after: updated
    });

    // 3.3 发布事件 (WebSocket 推送)
    await this.eventBus.publish({
      type: 'profile_updated',
      userId,
      data: { updateType, changes: validated }
    });

    return updated;
  });
}
```

## 四、数据验证措施

### 措施 1: 前端安全解析

**工具**: `shared/contracts/validation.ts:parseJSONSafe()`

```typescript
export function parseJSONSafe<T>(
  json: string | null | undefined,
  onError?: (error: z.ZodError) => void
): T | null {
  if (!json) return null;

  try {
    return JSON.parse(json) as T;
  } catch (error) {
    if (process.env.NODE_ENV === 'development') {
      throw new ValidationError(`JSON 解析失败: ${error}`);
    }
    onError?.(error);
    logger.error('JSON 解析失败', { error, json });
    return null;
  }
}
```

### 措施 2: API 层验证

**工具**: `backend/src/controllers/validation/apiValidator.ts:validateOrThrow()`

```typescript
export function validateOrThrow<T>(
  schema: z.ZodSchema<T>,
  data: unknown
): T {
  const result = schema.safeParse(data);

  if (!result.success) {
    logger.error('验证失败', {
      errors: result.error.errors,
      data
    });
    throw new ValidationError('数据验证失败', result.error.errors);
  }

  return result.data;
}
```

### 措施 3: 业务规则验证

**服务**: `backend/src/services/mas/services/ValidationService.ts`

```typescript
export class ValidationService {
  // 自定义验证器
  private loadAnchorRefiners = {
    severityMatchesExpiration: (data: any) => {
      if (data.severity >= 8 && !data.expire_at) {
        throw new BusinessRuleError('High severity requires expiration');
      }
    },

    noPermanentInjuryConflict: (data: any) => {
      if (data.part === 'permanent' && data.expire_at) {
        throw new BusinessRuleError('Permanent injury cannot expire');
      }
    }
  };

  async validateLoadAnchor(data: unknown): Promise<LoadAnchor> {
    // 先结构验证
    const parsed = LoadAnchorSchema.parse(data);

    // 再业务规则验证
    Object.values(this.loadAnchorRefiners).forEach(refiner => {
      refiner(parsed);
    });

    return parsed;
  }
}
```

## 六、故障排查

### 问题 1: 前端显示数据未更新

**检查点**:
1. WebSocket 连接状态 (`ws.readyState`)
2. 网络面板查看 `profile_updated` 事件
3. 查询缓存失效: `queryClient.invalidateQueries()`

### 问题 2: 数据库更新未生效

**检查点**:
1. 事务提交状态 (`tx.commit`)
2. 审计日志确认 (`audit_logs` 表)
3. PostgreSQL WAL 日志

### 问题 3: 数据格式验证失败

**检查点**:
1. 开发环境: 查看抛出的 ValidationError
2. 生产环境: 查看日志记录的 `validation_error`
3. LangSmith: 查看 LLM 结构化输出
